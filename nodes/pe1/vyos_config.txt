# pe1: PE in dc1 — day-0 pushed over the serial console by lab.sh bootstrap; rendered by tools/render.py
set system host-name pe1
set system domain-name lab.local
set system login user vyos authentication plaintext-password vyos
set service ssh port 22
set service lldp interface all
set interfaces ethernet eth0 address 10.3.0.11/24
set interfaces ethernet eth0 description 'OOB management'
set protocols static route 10.0.0.0/8 next-hop 10.3.0.1
# monitoring: Prometheus exporters on the OOB address (node-exporter :9100, frr-exporter :9342), scraped from the NMS
set service monitoring prometheus node-exporter listen-address 10.3.0.11
set service monitoring prometheus frr-exporter listen-address 10.3.0.11
# telemetry pushed by the node itself: VyOS Telegraf -> VictoriaMetrics on the NMS over the InfluxDB v2 write API (host
# metrics, VyOS service state, kernel nstat counters), every series tagged lab / role / dc; syslog -> VictoriaLogs (UDP 5514)
set service monitoring telegraf influxdb url http://10.3.0.10
set service monitoring telegraf influxdb port 8428
set service monitoring telegraf influxdb bucket srv6-core
set service monitoring telegraf influxdb authentication organization lab
set service monitoring telegraf influxdb authentication token srv6core-lab-telegraf_________________________________________________________________==
set system syslog remote 10.3.0.10 port 5514
set system syslog remote 10.3.0.10 protocol udp
set system syslog remote 10.3.0.10 facility all level info
set service monitoring telegraf global-tag lab value srv6-core
set service monitoring telegraf global-tag role value pe
set service monitoring telegraf global-tag dc value dc1
# underlay: IPv6-only core, IS-IS level-2 point-to-point, jumbo frames leave headroom for the SRv6 encapsulation
set interfaces loopback lo address fd00:a::1/128
set interfaces ethernet eth1 address fd00:b:0:101::2/64
set interfaces ethernet eth1 description 'core: p1 eth3'
set interfaces ethernet eth1 mtu 9000
set interfaces ethernet eth2 address fd00:b:0:201::2/64
set interfaces ethernet eth2 description 'core: p2 eth3'
set interfaces ethernet eth2 mtu 9000
# SRv6: the locator (usid-f3216: block 32 / node 16 / function 16 bits); dum0 is where FRR installs
# the local SIDs — addressed with a /128 so the connected route never outranks the locator's own End SID in zebra
set interfaces dummy dum0 address fd00:c:1::1/128
set interfaces dummy dum0 description 'SRv6 locator fd00:c:1::/48 (local SIDs)'
set protocols segment-routing srv6 locator main prefix fd00:c:1::/48
set protocols segment-routing srv6 locator main block-len 32
set protocols segment-routing srv6 locator main node-len 16
set protocols segment-routing srv6 locator main func-bits 16
set protocols segment-routing srv6 locator main format usid-f3216
set protocols segment-routing srv6 locator main behavior-usid
set protocols segment-routing srv6 encapsulation source-address fd00:a::1
set protocols segment-routing interface eth1 srv6
set protocols segment-routing interface eth2 srv6
set system sysctl parameter net.ipv6.conf.all.seg6_enabled value 1
set protocols isis net 49.0001.0000.0000.0001.00
set protocols isis level level-2
set protocols isis metric-style wide
set protocols isis log-adjacency-changes
set protocols isis interface eth1 network point-to-point
set protocols isis interface eth2 network point-to-point
set protocols isis interface eth1 bfd
set protocols isis interface eth2 bfd
set protocols isis interface lo passive
set protocols isis interface dum0 passive
set protocols isis segment-routing srv6 locator main
set protocols isis segment-routing srv6 interface dum0
# BGP: VPNv4 to the route reflectors p1, p3 over the IPv6 loopbacks (extended next hop), SRv6 SIDs from locator main
set protocols bgp system-as 65000
set protocols bgp parameters router-id 10.255.0.1
set protocols bgp parameters log-neighbor-changes
set protocols bgp srv6 locator main
set protocols bgp neighbor fd00:a::11 remote-as 65000
set protocols bgp neighbor fd00:a::11 description 'p1 route reflector'
set protocols bgp neighbor fd00:a::11 update-source fd00:a::1
set protocols bgp neighbor fd00:a::11 capability extended-nexthop
set protocols bgp neighbor fd00:a::11 address-family ipv4-vpn
set protocols bgp neighbor fd00:a::13 remote-as 65000
set protocols bgp neighbor fd00:a::13 description 'p3 route reflector'
set protocols bgp neighbor fd00:a::13 update-source fd00:a::1
set protocols bgp neighbor fd00:a::13 capability extended-nexthop
set protocols bgp neighbor fd00:a::13 address-family ipv4-vpn
# tenant VRF tenant-a (table 100, RT 65000:100, RD 65000:101): attachment circuit eth3 to ce1 eth1
set vrf name tenant-a table 100
set interfaces ethernet eth3 vrf tenant-a
set interfaces ethernet eth3 address 172.16.1.1/30
set interfaces ethernet eth3 description 'tenant-a: ce1 eth1'
# Linux scopes the SRv6 encapsulation's outer lookup to the ingress VRF for forwarded packets: leak every remote locator into
# the VRF table via the attached P router(s) on the IGP shortest path (recursive through IS-IS, so a dead P drops out), and the
# whole block via every attached P as the fallback
set vrf name tenant-a protocols static route6 fd00:c:11::/48 next-hop fd00:a::11 vrf default
set vrf name tenant-a protocols static route6 fd00:c:12::/48 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c:13::/48 next-hop fd00:a::11 vrf default
set vrf name tenant-a protocols static route6 fd00:c:13::/48 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c:2::/48 next-hop fd00:a::11 vrf default
set vrf name tenant-a protocols static route6 fd00:c:2::/48 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c:3::/48 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c:4::/48 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c::/32 next-hop fd00:a::11 vrf default
set vrf name tenant-a protocols static route6 fd00:c::/32 next-hop fd00:a::12 vrf default
# eBGP to the CE; export/import with an SRv6 End.DT4 SID (sid vpn export auto)
set vrf name tenant-a protocols bgp system-as 65000
set vrf name tenant-a protocols bgp parameters router-id 10.255.0.1
set vrf name tenant-a protocols bgp parameters log-neighbor-changes
set vrf name tenant-a protocols bgp neighbor 172.16.1.2 remote-as 65001
set vrf name tenant-a protocols bgp neighbor 172.16.1.2 description 'ce1 (tenant-a)'
set vrf name tenant-a protocols bgp neighbor 172.16.1.2 address-family ipv4-unicast
set vrf name tenant-a protocols bgp address-family ipv4-unicast redistribute connected
set vrf name tenant-a protocols bgp address-family ipv4-unicast rd vpn export 65000:101
set vrf name tenant-a protocols bgp address-family ipv4-unicast route-target vpn both 65000:100
set vrf name tenant-a protocols bgp address-family ipv4-unicast sid vpn export auto
set vrf name tenant-a protocols bgp address-family ipv4-unicast import vpn
set vrf name tenant-a protocols bgp address-family ipv4-unicast export vpn

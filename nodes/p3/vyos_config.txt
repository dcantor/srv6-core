# p3: P in core — day-0 pushed over the serial console by lab.sh bootstrap; rendered by tools/render.py
set system host-name p3
set system domain-name lab.local
set system login user vyos authentication plaintext-password vyos
set service ssh port 22
set service lldp interface all
set interfaces ethernet eth0 address 10.3.0.23/24
set interfaces ethernet eth0 description 'OOB management'
set protocols static route 10.0.0.0/8 next-hop 10.3.0.1
# underlay: IPv6-only core, IS-IS level-2 point-to-point, jumbo frames leave headroom for the SRv6 encapsulation
set interfaces loopback lo address fd00:a::13/128
set interfaces ethernet eth1 address fd00:b:0:13::2/64
set interfaces ethernet eth1 description 'core: p1 eth2'
set interfaces ethernet eth1 mtu 9000
set interfaces ethernet eth2 address fd00:b:0:23::2/64
set interfaces ethernet eth2 description 'core: p2 eth2'
set interfaces ethernet eth2 mtu 9000
set interfaces ethernet eth3 address fd00:b:0:303::1/64
set interfaces ethernet eth3 description 'core: pe3 eth2'
set interfaces ethernet eth3 mtu 9000
set interfaces ethernet eth4 address fd00:b:0:304::1/64
set interfaces ethernet eth4 description 'core: pe4 eth2'
set interfaces ethernet eth4 mtu 9000
# SRv6: the locator; dum0 carries the local SIDs (FRR installs them there) and, advertised passively by IS-IS, keeps the
# locator reachable even when the IS-IS SRv6 sub-TLVs are not understood by a neighbour
set interfaces dummy dum0 address fd00:c:13::1/64
set interfaces dummy dum0 description 'SRv6 locator fd00:c:13::/64 (local SIDs)'
set protocols segment-routing srv6 locator main prefix fd00:c:13::/64
set protocols segment-routing srv6 locator main block-len 40
set protocols segment-routing srv6 locator main node-len 24
set protocols segment-routing srv6 locator main func-bits 16
set protocols segment-routing srv6 encapsulation source-address fd00:a::13
set protocols segment-routing interface eth1 srv6
set protocols segment-routing interface eth2 srv6
set protocols segment-routing interface eth3 srv6
set protocols segment-routing interface eth4 srv6
set system sysctl parameter net.ipv6.conf.all.seg6_enabled value 1
set protocols isis net 49.0001.0000.0000.0013.00
set protocols isis level level-2
set protocols isis metric-style wide
set protocols isis log-adjacency-changes
set protocols isis interface eth1 network point-to-point
set protocols isis interface eth2 network point-to-point
set protocols isis interface eth3 network point-to-point
set protocols isis interface eth4 network point-to-point
set protocols isis interface eth1 bfd
set protocols isis interface eth2 bfd
set protocols isis interface eth3 bfd
set protocols isis interface eth4 bfd
set protocols isis interface lo passive
set protocols isis interface dum0 passive
set protocols isis segment-routing srv6 locator main
set protocols isis segment-routing srv6 interface dum0
# VPNv4 route reflector for the PEs (no VRFs here; p routers only forward IPv6); the PEs peer with every reflector
set protocols bgp system-as 65000
set protocols bgp parameters router-id 10.255.0.13
set protocols bgp parameters cluster-id 10.255.0.13
set protocols bgp parameters log-neighbor-changes
set protocols bgp peer-group RR-CLIENTS remote-as 65000
set protocols bgp peer-group RR-CLIENTS update-source fd00:a::13
set protocols bgp peer-group RR-CLIENTS capability extended-nexthop
set protocols bgp peer-group RR-CLIENTS address-family ipv4-vpn route-reflector-client
set protocols bgp neighbor fd00:a::1 peer-group RR-CLIENTS
set protocols bgp neighbor fd00:a::1 description 'pe1'
set protocols bgp neighbor fd00:a::2 peer-group RR-CLIENTS
set protocols bgp neighbor fd00:a::2 description 'pe2'
set protocols bgp neighbor fd00:a::3 peer-group RR-CLIENTS
set protocols bgp neighbor fd00:a::3 description 'pe3'
set protocols bgp neighbor fd00:a::4 peer-group RR-CLIENTS
set protocols bgp neighbor fd00:a::4 description 'pe4'

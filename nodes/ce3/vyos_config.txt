# ce3: CE in dc3 — day-0 pushed over the serial console by lab.sh bootstrap; rendered by tools/gen_configs.py
set system host-name ce3
set system domain-name lab.local
set system login user vyos authentication plaintext-password vyos
set service ssh port 22
set service lldp interface all
set interfaces ethernet eth0 address 10.3.0.33/24
set interfaces ethernet eth0 description 'OOB management'
set protocols static route 10.0.0.0/8 next-hop 10.3.0.1
# attachment circuit to pe3 (VRF tenant-a on the PE side) and the dc3 LAN
set interfaces ethernet eth1 address 172.16.3.2/30
set interfaces ethernet eth1 description 'pe3 eth3 (tenant-a)'
set interfaces ethernet eth2 address 172.20.3.1/24
set interfaces ethernet eth2 description 'dc3 LAN: h3'
# eBGP to the PE: announce the LAN, learn the other data centres
set protocols bgp system-as 65003
set protocols bgp parameters router-id 172.20.3.1
set protocols bgp neighbor 172.16.3.1 remote-as 65000
set protocols bgp neighbor 172.16.3.1 description 'pe3'
set protocols bgp neighbor 172.16.3.1 address-family ipv4-unicast
set protocols bgp address-family ipv4-unicast network 172.20.3.0/24

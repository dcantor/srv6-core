# ce2: CE in dc2 — day-0 pushed over the serial console by lab.sh bootstrap; rendered by tools/render.py
set system host-name ce2
set system domain-name lab.local
set system login user vyos authentication plaintext-password vyos
set service ssh port 22
set service lldp interface all
set interfaces ethernet eth0 address 10.3.0.32/24
set interfaces ethernet eth0 description 'OOB management'
set protocols static route 10.0.0.0/8 next-hop 10.3.0.1
# monitoring: Prometheus exporters on the OOB address (node-exporter :9100, frr-exporter :9342), scraped from the NMS
set service monitoring prometheus node-exporter listen-address 10.3.0.32
set service monitoring prometheus frr-exporter listen-address 10.3.0.32
# telemetry pushed by the node itself: VyOS Telegraf -> VictoriaMetrics on the NMS over the InfluxDB v2 write API (host
# metrics, VyOS service state, kernel nstat counters), every series tagged lab / role / dc; syslog -> VictoriaLogs (UDP 5514)
set service monitoring telegraf influxdb url http://10.3.0.10
set service monitoring telegraf influxdb port 8428
set service monitoring telegraf influxdb bucket srv6-core
set service monitoring telegraf influxdb authentication organization lab
set service monitoring telegraf influxdb authentication token srv6core-lab-telegraf_________________________________________________________________==
set system syslog remote 10.3.0.10 port 5514
set system syslog remote 10.3.0.10 protocol udp
set system syslog remote 10.3.0.10 facility all level info
set service monitoring telegraf global-tag lab value srv6-core
set service monitoring telegraf global-tag role value ce
set service monitoring telegraf global-tag dc value dc2
# VyOS/FRR insist on a default-VRF BGP instance while VRF instances exist: an empty one (no neighbours, no networks)
set protocols bgp system-as 65002
# tenant-a: VRF tenant-a on the CE holds the attachment circuit to pe2 and the dc2 LAN (dual-stack: one eBGP session per family)
set vrf name tenant-a table 100
set interfaces ethernet eth1 vrf tenant-a
set interfaces ethernet eth2 vrf tenant-a
set interfaces ethernet eth1 address 172.16.2.2/30
set interfaces ethernet eth1 address fd00:16:2::2/64
set interfaces ethernet eth1 description 'pe2 eth3 (tenant-a)'
set interfaces ethernet eth2 address 172.20.2.1/24
set interfaces ethernet eth2 address fd00:20:2::1/64
set interfaces ethernet eth2 description 'dc2 LAN tenant-a: dc2-h1'
set vrf name tenant-a protocols bgp system-as 65002
set vrf name tenant-a protocols bgp parameters router-id 172.20.2.1
set vrf name tenant-a protocols bgp parameters log-neighbor-changes
set vrf name tenant-a protocols bgp neighbor 172.16.2.1 remote-as 65000
set vrf name tenant-a protocols bgp neighbor 172.16.2.1 description 'pe2 (tenant-a)'
set vrf name tenant-a protocols bgp neighbor 172.16.2.1 address-family ipv4-unicast
set vrf name tenant-a protocols bgp address-family ipv4-unicast network 172.20.2.0/24
set vrf name tenant-a protocols bgp neighbor fd00:16:2::1 remote-as 65000
set vrf name tenant-a protocols bgp neighbor fd00:16:2::1 description 'pe2 (tenant-a, IPv6)'
set vrf name tenant-a protocols bgp neighbor fd00:16:2::1 address-family ipv6-unicast
set vrf name tenant-a protocols bgp address-family ipv6-unicast network fd00:20:2::/64
# tenant-b: VRF tenant-b on the CE holds the attachment circuit to pe2 and the dc2 LAN (dual-stack: one eBGP session per family)
set vrf name tenant-b table 200
set interfaces ethernet eth3 vrf tenant-b
set interfaces ethernet eth4 vrf tenant-b
set interfaces ethernet eth3 address 172.18.2.2/30
set interfaces ethernet eth3 address fd00:18:2::2/64
set interfaces ethernet eth3 description 'pe2 eth4 (tenant-b)'
set interfaces ethernet eth4 address 172.21.2.1/24
set interfaces ethernet eth4 address fd00:21:2::1/64
set interfaces ethernet eth4 description 'dc2 LAN tenant-b: dc2-h2'
set vrf name tenant-b protocols bgp system-as 65002
set vrf name tenant-b protocols bgp parameters router-id 172.21.2.1
set vrf name tenant-b protocols bgp parameters log-neighbor-changes
set vrf name tenant-b protocols bgp neighbor 172.18.2.1 remote-as 65000
set vrf name tenant-b protocols bgp neighbor 172.18.2.1 description 'pe2 (tenant-b)'
set vrf name tenant-b protocols bgp neighbor 172.18.2.1 address-family ipv4-unicast
set vrf name tenant-b protocols bgp address-family ipv4-unicast network 172.21.2.0/24
set vrf name tenant-b protocols bgp neighbor fd00:18:2::1 remote-as 65000
set vrf name tenant-b protocols bgp neighbor fd00:18:2::1 description 'pe2 (tenant-b, IPv6)'
set vrf name tenant-b protocols bgp neighbor fd00:18:2::1 address-family ipv6-unicast

# pe3: PE in dc3 — day-0 pushed over the serial console by lab.sh bootstrap; rendered by tools/gen_configs.py
set system host-name pe3
set system domain-name lab.local
set system login user vyos authentication plaintext-password vyos
set service ssh port 22
set service lldp interface all
set interfaces ethernet eth0 address 10.3.0.13/24
set interfaces ethernet eth0 description 'OOB management'
set protocols static route 10.0.0.0/8 next-hop 10.3.0.1
# underlay: IPv6-only core, IS-IS level-2 point-to-point, jumbo frames leave headroom for the SRv6 encapsulation
set interfaces loopback lo address fd00:a::3/128
set interfaces ethernet eth1 address fd00:b:0:203::2/64
set interfaces ethernet eth1 description 'core: p2 eth5'
set interfaces ethernet eth1 mtu 9000
set interfaces ethernet eth2 address fd00:b:0:303::2/64
set interfaces ethernet eth2 description 'core: p3 eth3'
set interfaces ethernet eth2 mtu 9000
# SRv6: the locator; dum0 carries the local SIDs (FRR installs them there) and, advertised passively by IS-IS, keeps the
# locator reachable even when the IS-IS SRv6 sub-TLVs are not understood by a neighbour
set interfaces dummy dum0 address fd00:c:3::1/64
set interfaces dummy dum0 description 'SRv6 locator fd00:c:3::/64 (local SIDs)'
set protocols segment-routing srv6 locator main prefix fd00:c:3::/64
set protocols segment-routing srv6 locator main block-len 40
set protocols segment-routing srv6 locator main node-len 24
set protocols segment-routing srv6 locator main func-bits 16
set protocols segment-routing srv6 encapsulation source-address fd00:a::3
set protocols segment-routing interface eth1 srv6
set protocols segment-routing interface eth2 srv6
set system sysctl parameter net.ipv6.conf.all.seg6_enabled value 1
set protocols isis net 49.0001.0000.0000.0003.00
set protocols isis level level-2
set protocols isis metric-style wide
set protocols isis log-adjacency-changes
set protocols isis interface eth1 network point-to-point
set protocols isis interface eth2 network point-to-point
set protocols isis interface lo passive
set protocols isis interface dum0 passive
set protocols isis segment-routing srv6 locator main
set protocols isis segment-routing srv6 interface dum0
# customer VRF tenant-a: attachment circuit to ce3
set vrf name tenant-a table 100
set interfaces ethernet eth3 vrf tenant-a
set interfaces ethernet eth3 address 172.16.3.1/30
set interfaces ethernet eth3 description 'tenant-a: ce3 eth1'
# Linux scopes the SRv6 encapsulation's outer lookup to the ingress VRF for forwarded packets: leak the locator block into the
# VRF table, recursively via the attached P routers' loopbacks (resolved by IS-IS, so ECMP and failover are kept)
set vrf name tenant-a protocols static route6 fd00:c::/40 next-hop fd00:a::12 vrf default
set vrf name tenant-a protocols static route6 fd00:c::/40 next-hop fd00:a::13 vrf default
# BGP: VPNv4 to the route reflector p1 over the IPv6 loopbacks (extended next hop), SRv6 SIDs from locator main
set protocols bgp system-as 65000
set protocols bgp parameters router-id 10.255.0.3
set protocols bgp parameters log-neighbor-changes
set protocols bgp srv6 locator main
set protocols bgp neighbor fd00:a::11 remote-as 65000
set protocols bgp neighbor fd00:a::11 description 'p1 route reflector'
set protocols bgp neighbor fd00:a::11 update-source fd00:a::3
set protocols bgp neighbor fd00:a::11 capability extended-nexthop
set protocols bgp neighbor fd00:a::11 address-family ipv4-vpn
# BGP in the VRF: eBGP to the CE; export/import with an SRv6 End.DT4 SID (sid vpn export auto)
set vrf name tenant-a protocols bgp system-as 65000
set vrf name tenant-a protocols bgp parameters router-id 10.255.0.3
set vrf name tenant-a protocols bgp neighbor 172.16.3.2 remote-as 65003
set vrf name tenant-a protocols bgp neighbor 172.16.3.2 description 'ce3'
set vrf name tenant-a protocols bgp neighbor 172.16.3.2 address-family ipv4-unicast
set vrf name tenant-a protocols bgp address-family ipv4-unicast redistribute connected
set vrf name tenant-a protocols bgp address-family ipv4-unicast rd vpn export 65000:3
set vrf name tenant-a protocols bgp address-family ipv4-unicast route-target vpn both 65000:100
set vrf name tenant-a protocols bgp address-family ipv4-unicast sid vpn export auto
set vrf name tenant-a protocols bgp address-family ipv4-unicast import vpn
set vrf name tenant-a protocols bgp address-family ipv4-unicast export vpn
